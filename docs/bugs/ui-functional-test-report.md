let srv = null;
async function up(timeoutMs = 30000) {
  const t0 = Date.now();
  while (Date.now() - t0 < timeoutMs) {
    try { const r = await fetch(BASE + '/api/system'); if (r.ok) return true; } catch { /* not yet */ }
    await sleep(250);
  }
  throw new Error('server did not come up at ' + BASE);
}
async function down(timeoutMs = 15000) {
  const t0 = Date.now();
  while (Date.now() - t0 < timeoutMs) {
    try { await fetch(BASE + '/api/system'); } catch { return true; }
    await sleep(200);
  }
  throw new Error('server did not stop');
}
async function startServer() {
  if (!SERVER_CMD) return;
  const log = fs.openSync(path.join(OUT_ROOT, 'server.log'), 'a');
  srv = cp.spawn(SERVER_CMD, { shell: true, detached: true, stdio: ['ignore', log, log] });
  await up();
}
async function stopServer() {
  if (!SERVER_CMD) return;
  if (KILL_CMD) cp.execSync(KILL_CMD, { stdio: 'ignore' });
  else if (srv) { try { process.kill(-srv.pid, 'SIGTERM'); } catch { /* gone */ } }
  srv = null;
  await down();
}
class Blocked extends Error {}
async function restartServer() {
  if (!SERVER_CMD) throw new Blocked('no SERVER_CMD — the suite cannot restart the server');
  await stopServer();
  await startServer();
}

function crc32(buf) {
  let c, crc = 0xFFFFFFFF;
  for (let n = 0; n < buf.length; n++) {
    c = (crc ^ buf[n]) & 0xFF;
    for (let k = 0; k < 8; k++) c = c & 1 ? 0xEDB88320 ^ (c >>> 1) : c >>> 1;
    crc = (crc >>> 8) ^ c;
  }
  return (crc ^ 0xFFFFFFFF) >>> 0;
}
function writeZip(file, entries /* [{name, data:Buffer}] */) {
  const parts = [], central = []; let off = 0;
  for (const e of entries) {
    const name = Buffer.from(e.name), crc = crc32(e.data), sz = e.data.length;
    const lh = Buffer.alloc(30); lh.writeUInt32LE(0x04034b50, 0); lh.writeUInt16LE(20, 4);
    lh.writeUInt32LE(crc, 14); lh.writeUInt32LE(sz, 18); lh.writeUInt32LE(sz, 22); lh.writeUInt16LE(name.length, 26);
    parts.push(lh, name, e.data);
    const ch = Buffer.alloc(46); ch.writeUInt32LE(0x02014b50, 0); ch.writeUInt16LE(20, 4); ch.writeUInt16LE(20, 6);
    ch.writeUInt32LE(crc, 16); ch.writeUInt32LE(sz, 20); ch.writeUInt32LE(sz, 24); ch.writeUInt16LE(name.length, 28);
    ch.writeUInt32LE(off, 42);
    central.push(ch, name);
    off += 30 + name.length + sz;
  }
  const cd = Buffer.concat(central);
  const end = Buffer.alloc(22); end.writeUInt32LE(0x06054b50, 0); end.writeUInt16LE(entries.length, 8);
  end.writeUInt16LE(entries.length, 10); end.writeUInt32LE(cd.length, 12); end.writeUInt32LE(off, 16);
  fs.writeFileSync(file, Buffer.concat([...parts, cd, end]));
}

async function openApp(opts = {}) {
  const ctx = await browser.newContext({ viewport: opts.viewport || { width: 960, height: 640 }, deviceScaleFactor: 1 });
  current.contexts.push(ctx);
  await ctx.route(u => !u.href.startsWith(BASE), r => r.abort()); // never leave localhost

  page._dialogs = []; page._req = [];

  page.on('console', m => { if (m.type() === 'error') ERRORS.push({ test: current.id, kind: 'console.error', text: m.text() }); });
  page.on('pageerror', e => ERRORS.push({ test: current.id, kind: 'pageerror', text: e.message }));

  if (opts.routes) await opts.routes(page);

  if (!opts.noWait) await page.waitForFunction(() => document.getElementById('combo-ver').textContent.length > 0, null, { timeout: 15000 });
  return page;
}

async function layoutIssues(p) {
  return p.evaluate(() => {
    const d = e => (e.id ? '#' + e.id : e.tagName.toLowerCase() + (e.className && typeof e.className === 'string' ? '.' + e.className.trim().split(/\s+/).join('.') : '')) + (e.textContent.trim() ? `("${e.textContent.trim().slice(0, 24)}")` : '');
    const out = [];
    const sel = 'button, input, a, .acard, .combo-field, .opt, .opt-head, .switch-row, .seg, .list, .log, .banner, .bbtns, .panes, .detect, .dl-card, .out-row, .left-head, .list-foot, .warn-banner, .seg-row';
    for (const el of document.querySelectorAll(sel)) {
      if (!el.offsetParent || getComputedStyle(el).visibility === 'hidden') continue;
      const r = el.getBoundingClientRect(); if (!r.width || !r.height) continue;
      for (let a = el.parentElement; a; a = a.parentElement) {
        const cs = getComputedStyle(a); if (cs.overflowX === 'visible' && cs.overflowY === 'visible') continue;
        const ar = a.getBoundingClientRect();
        const outside = r.left < ar.left - 1 || r.right > ar.right + 1 || r.top < ar.top - 1 || r.bottom > ar.bottom + 1;
        if (!outside) continue;
        if (/(auto|scroll)/.test(cs.overflowY + cs.overflowX)) break; // reachable by scrolling that container
        out.push(`${d(el)} [${[r.left, r.top, r.right, r.bottom].map(Math.round)}] clipped by ${d(a).slice(0, 30)} [${[ar.left, ar.top, ar.right, ar.bottom].map(Math.round)}]`);
        break;
      }
    }
    for (const el of document.querySelectorAll('.cname, .tname, .opt-title, .at, .ad, .bt, .rp-title, .step, .dl-sub, .run-title'))
      if (el.offsetParent && el.scrollWidth > el.clientWidth + 1) out.push(`text overflow ${d(el)}`);
    if (document.scrollingElement.scrollWidth > innerWidth) out.push(`horizontal page scroll ${document.scrollingElement.scrollWidth}>${innerWidth}`);
    return out;
  });
}
